這個範例包含了實現字幕模式的所有關鍵技術點：

動態定位：SubtitleWindow.cs 中的 UpdatePositionAndSize 方法會監聽主視窗的位置和尺寸變化，確保字幕視窗始終精確地依附在主工具條下方。

視覺統一：SubtitleWindow.xaml 的 Border 使用了 CornerRadius="0,0,26,26"，使其底部圓角能與主工具條的頂部圓角形成一個完美的視覺整體。

動畫效果：新的字幕行出現時，會有一個平滑的淡入動畫，提升了視覺體驗。

自動滾動：當字幕行數超過顯示區域時，ScrollViewer 會自動滾動到最新的一行。

MVVM 架構：程式碼遵循了 MVVM 設計模式，UI (View) 和資料 (ViewModel) 之間是解耦的，透過命令和資料綁定進行互動。

您可以將這份程式碼作為您 MonLingo 專案中字幕模式的基礎，它已經具備了 Gaminik 的核心設計精髓。

        public MainBarWindow()

                    xmlns:x='http://schemas.microsoft.com/winfx/2006/xaml'

    // ============================================================================================
    // 5. SubtitleWindow.cs - 字幕視窗後端邏輯
    // ============================================================================================
    public partial class SubtitleWindow : Window
    {
        private readonly Window _ownerWindow;
        private ScrollViewer _scrollViewer;

        public SubtitleWindow(Window owner)
        {
            _ownerWindow = owner;
            this.Owner = owner; // 建立擁有者關係

            var xaml = @"
            <Window xmlns='http://schemas.microsoft.com/winfx/2006/xaml/presentation'
                    xmlns:x='http://schemas.microsoft.com/winfx/2006/xaml'
                    x:Class='MonLingoSubtitleDemo.SubtitleWindow'
                    Title='SubtitleWindow' Height='120'
                    WindowStyle='None' AllowsTransparency='True' Background='Transparent'
                    Topmost='True' ShowInTaskbar='False' IsHitTestVisible='False'>
                <Border CornerRadius='0,0,26,26' Background='#AA333333'>
                    <ScrollViewer x:Name='PART_ScrollViewer' VerticalScrollBarVisibility='Hidden' HorizontalScrollBarVisibility='Hidden'>
                        <ItemsControl ItemsSource='{Binding SubtitleLines}'>
                            <ItemsControl.ItemTemplate>
                                <DataTemplate>
                                    <Grid>
                                        <Grid.Style>
                                            <Style TargetType='Grid'>
                                                <Setter Property='Opacity' Value='0'/>
                                                <Style.Triggers>
                                                    <EventTrigger RoutedEvent='Loaded'>
                                                        <BeginStoryboard>
                                                            <Storyboard>
                                                                <DoubleAnimation Storyboard.TargetProperty='Opacity' From='0' To='1' Duration='0:0:0.5'/>
                                                            </Storyboard>
                                                        </BeginStoryboard>
                                                    </EventTrigger>
                                                </Style.Triggers>
                                            </Style>
                                        </Grid.Style>
                                        <TextBlock Text='{Binding}' Foreground='White' FontSize='18' Margin='20,5' TextWrapping='Wrap'/>
                                    </Grid>
                                </DataTemplate>
                            </ItemsControl.ItemTemplate>
                        </ItemsControl>
                    </ScrollViewer>
                </Border>
            </Window>";
            
            var window = (Window)XamlReader.Parse(xaml);
            this.Content = window.Content;
            this.Height = window.Height;
            this.WindowStyle = window.WindowStyle;
            this.AllowsTransparency = window.AllowsTransparency;
            this.Background = window.Background;
            this.Topmost = window.Topmost;
            this.ShowInTaskbar = window.ShowInTaskbar;
            this.IsHitTestVisible = window.IsHitTestVisible;

            this.Loaded += SubtitleWindow_Loaded;
        }

        private void SubtitleWindow_Loaded(object sender, RoutedEventArgs e)
        {
            // 訂閱主視窗的位置和尺寸變化事件
            _ownerWindow.LocationChanged += OwnerWindow_LocationChanged;
            _ownerWindow.SizeChanged += OwnerWindow_SizeChanged;
            _ownerWindow.StateChanged += OwnerWindow_StateChanged;
            _ownerWindow.IsVisibleChanged += OwnerWindow_IsVisibleChanged;

            // 獲取 ScrollViewer 的引用
            _scrollViewer = FindVisualChild<ScrollViewer>(this);

            // 訂閱 ViewModel 的事件以實現自動滾動
            if (this.DataContext is SubtitleViewModel vm)
            {
                vm.NewLineAdded += OnNewLineAdded;
            }
            
            // 初始定位
            UpdatePositionAndSize();
        }

        private void OwnerWindow_IsVisibleChanged(object sender, DependencyPropertyChangedEventArgs e)
        {
            if(this.DataContext is SubtitleViewModel vm && vm.IsSubtitleVisible)
            {
                this.Visibility = _ownerWindow.IsVisible ? Visibility.Visible : Visibility.Collapsed;
            }
        }

        private void OwnerWindow_StateChanged(object sender, EventArgs e)
        {
            if (_ownerWindow.WindowState == WindowState.Minimized)
            {
                this.Visibility = Visibility.Collapsed;
            }
            else if (this.DataContext is SubtitleViewModel vm && vm.IsSubtitleVisible)
            {
                this.Visibility = Visibility.Visible;
            }
        }

        private void OnNewLineAdded()
        {
            // 當 ViewModel 新增一行時，自動滾動到底部
            _scrollViewer?.ScrollToEnd();
        }

        private void OwnerWindow_LocationChanged(object sender, EventArgs e) => UpdatePositionAndSize();
        private void OwnerWindow_SizeChanged(object sender, SizeChangedEventArgs e) => UpdatePositionAndSize();

        private void UpdatePositionAndSize()
        {
            // 【關鍵】動態更新字幕視窗的位置和寬度，使其依附於主工具條
            this.Width = _ownerWindow.ActualWidth;
            this.Left = _ownerWindow.Left;
            this.Top = _ownerWindow.Top + _ownerWindow.ActualHeight;
        }

        // 輔助方法，用於在視覺化樹中找到指定的子控制項
        private T FindVisualChild<T>(DependencyObject parent) where T : DependencyObject
        {
            for (int i = 0; i < VisualTreeHelper.GetChildrenCount(parent); i++)
            {
                var child = VisualTreeHelper.GetChild(parent, i);
                if (child is T t)
                {
                    return t;
                }
                var childOfChild = FindVisualChild<T>(child);
                if (childOfChild != null)
                {
                    return childOfChild;
                }
            }
            return null;
        }

        protected override void OnClosing(CancelEventArgs e)
        {
            // 清理事件訂閱，防止記憶體洩漏
            _ownerWindow.LocationChanged -= OwnerWindow_LocationChanged;
            _ownerWindow.SizeChanged -= OwnerWindow_SizeChanged;
            _ownerWindow.StateChanged -= OwnerWindow_StateChanged;
            _ownerWindow.IsVisibleChanged -= OwnerWindow_IsVisibleChanged;
            if (this.DataContext is SubtitleViewModel vm)
            {
                vm.NewLineAdded -= OnNewLineAdded;
            }
            base.OnClosing(e);
        }
    }


    // ============================================================================================
    // 6. SubtitleViewModel.cs - 字幕視窗的 ViewModel
    // ============================================================================================
    public partial class SubtitleViewModel : ObservableObject
    {
        private int _lineCounter = 0;

        [ObservableProperty]
        private bool _isSubtitleVisible = false;

        public ObservableCollection<string> SubtitleLines { get; } = new();

        private const int MaxLines = 5; // 最多顯示5行字幕

        // 由 TranslationPipelineManager 或主視窗的按鈕呼叫
        [RelayCommand]
        private void AddNewLine()
        {
            _lineCounter++;
            string newText = $"這是第 {_lineCounter} 行字幕。Gaminik 的字幕模式是依附於主工具條的。";
            
            if (SubtitleLines.Count >= MaxLines)
            {
                SubtitleLines.RemoveAt(0);
            }
            SubtitleLines.Add(newText);

            // 觸發事件通知 View 滾動
            NewLineAdded?.Invoke();
        }

        [RelayCommand]
        private void ToggleSubtitleVisibility()
        {
            IsSubtitleVisible = !IsSubtitleVisible;
            
            // 透過 Application.Current.Windows 找到對應的視窗並更新其可見性
            foreach (Window window in Application.Current.Windows)
            {
                if (window is SubtitleWindow subtitleWindow)
                {
                    subtitleWindow.Visibility = IsSubtitleVisible ? Visibility.Visible : Visibility.Collapsed;
                    if(IsSubtitleVisible)
                    {
                        // 確保它在主視窗之上
                        subtitleWindow.Topmost = false;
                        subtitleWindow.Owner.Activate();
                        subtitleWindow.Topmost = true;
                    }
                }
            }
        }
        
        public event Action NewLineAdded;
    }
